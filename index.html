train.ts 
import { Kasratshala } from '@kasratshala/sdk';

const k = new Kasratshala({
  apiKey: process.env.KASRATSHALA_KEY,
});

// provision a multi-GPU cluster
const cluster = await k.clusters.create({
  gpu:    'h100-sxm5',
  count:  8,
  region: 'us-east-1',
});

// launch a training job
const job = await k.jobs.run({
  clusterId: cluster.id,
  image:     'ghcr.io/acme/trainer:latest',
  command:   'python train.py --epochs 50',
});

// stream logs in real time
for await (const line of job.logs()) {
  console.log(line);
}
